cmake_minimum_required(VERSION 3.20)
project(chimeradb_wire_smoke C)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MONGOC QUIET IMPORTED_TARGET mongoc2)
if(NOT MONGOC_FOUND)
  pkg_check_modules(MONGOC REQUIRED IMPORTED_TARGET libmongoc-1.0)
endif()
add_executable(chimeradb-wire-smoke main.c)
target_link_libraries(chimeradb-wire-smoke PRIVATE PkgConfig::MONGOC)
set_target_properties(chimeradb-wire-smoke PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
